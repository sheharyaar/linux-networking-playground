## The measurement script behind the CSVs (`maglev_lab.py`, kept out of the lab folder)

```python
#!/usr/bin/env python3
"""maglev_lab.py: Maglev tables against a plain mod-N table, balance and disruption.
usage: python3 maglev_lab.py sweep  > sweep.csv    (N = 100, Cilium's table sizes)
       python3 maglev_lab.py fig12  > fig12.csv    (the paper's Fig. 12 setting, N = 1000)
"""
import random, sys

def populate(os_, M):                     # Pseudocode 1 (p. 6); os_ = [(offset, skip), ...]
    N = len(os_); nxt = [0] * N; entry = [-1] * M; n = 0
    while True:
        for i, (off, sk) in enumerate(os_):
            c = (off + nxt[i] * sk) % M
            while entry[c] >= 0:          # already taken: try the next preference
                nxt[i] += 1; c = (off + nxt[i] * sk) % M
            entry[c] = i; nxt[i] += 1; n += 1
            if n == M: return entry

def moved(before, after, dead):           # % of the table that changed owner, two ways
    extra = sum(1 for b, a in zip(before, after) if b != a and b not in dead)
    total = sum(1 for b, a in zip(before, after) if b != a)
    return 100 * extra / len(before), 100 * total / len(before)

def trial(M, N, k, rng):
    os_ = [(rng.randrange(M), rng.randrange(M - 1) + 1) for _ in range(N)]
    before = populate(os_, M)
    dead = set(rng.sample(range(N), k)); alive = [i for i in range(N) if i not in dead]
    after = [alive[x] for x in populate([os_[i] for i in alive], M)]
    cnt = [0] * N
    for b in before: cnt[b] += 1
    modn_b = [j % N for j in range(M)]                 # plain mod-N: slot j -> backend j mod N
    modn_a = [alive[j % len(alive)] for j in range(M)]
    return min(cnt), max(cnt), moved(before, after, dead), moved(modn_b, modn_a, dead)

def run(M, N, k, trials, rng):
    r = [trial(M, N, k, rng) for _ in range(trials)]
    avg = lambda f: sum(f(x) for x in r) / trials
    return (min(x[0] for x in r), max(x[1] for x in r), avg(lambda x: x[2][0]),
            avg(lambda x: x[2][1]), avg(lambda x: x[3][0]))

rng = random.Random(13)
if sys.argv[1:] == ['sweep']:
    print('M,N,M_over_N,k,trials,min_entries,max_entries,maglev_extra_pct,maglev_total_pct,modn_extra_pct')
    for M in (251, 509, 1021, 2039, 4093, 8191, 16381, 32749, 65521, 131071):  # Cilium's sizes
        for k in (1, 3):
            mn, mx, ex, tot, mod = run(M, 100, k, 20, rng)
            print(f'{M},100,{M/100:.2f},{k},20,{mn},{mx},{ex:.3f},{tot:.3f},{mod:.2f}', flush=True)
elif sys.argv[1:] == ['fig12']:
    print('M,N,k,failed_pct,trials,maglev_extra_pct,maglev_total_pct')
    for M, ks, t in ((65537, (1, 5, 10, 15, 20, 25, 30), 20), (655373, (1, 10, 20, 30), 3)):
        for k in ks:
            mn, mx, ex, tot, mod = run(M, 1000, k, t, rng)
            print(f'{M},1000,{k},{k/10:.1f},{t},{ex:.3f},{tot:.3f}', flush=True)
else:
    sys.exit(__doc__)
```
